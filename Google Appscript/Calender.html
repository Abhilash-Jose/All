Google Calendar Audit Tool

Complete Installation & Technical Code Guide

1. Instructions

Step 1: Open Google Sheets
Create a new blank sheet. Go to Extensions > Apps Script.
Step 2: Save the Script
Delete any placeholder code. Paste the block below into the editor. Click the Save icon and name it "CalendarAudit".
Step 3: Authorize
Refresh your Sheet. Click the new "My Audit Tools" menu. Follow the Google prompts to "Allow" access (use 'Advanced' settings if prompted).

2. The Commented Script

/** * 1. THE TRIGGER * This function creates a custom menu tab at the top of your Google Sheet * as soon as the file is opened or refreshed. */ function onOpen() { var ui = SpreadsheetApp.getUi(); ui.createMenu('My Audit Tools') .addItem('Audit Calendar (Export Events)', 'promptForCalendar') .addToUi(); } /** * 2. THE INPUT DIALOGUE * This function triggers a popup box to ask you how many days you want to scan. * It converts your text input into a number for the computer to process. */ function promptForCalendar() { var ui = SpreadsheetApp.getUi(); var response = ui.prompt('Calendar Audit', 'How many days into the future should I look?', ui.ButtonSet.OK_CANCEL); if (response.getSelectedButton() == ui.Button.OK) { var days = parseInt(response.getResponseText()); // Turn text into a number if (isNaN(days) || days < 1) { ui.alert('Please enter a valid number.'); } else { runCalendarAudit(days); // Pass that number to the main logic below } } } /** * 3. THE DATA EXPORTER * This is the "brain" of the tool. It clears your sheet, connects to * your calendar, calculates hours, and writes everything row-by-row. */ function runCalendarAudit(days) { var sheet = SpreadsheetApp.getActiveSheet(); sheet.clear(); // Wipe previous data to start fresh // Create bold headers for the top of the columns var headers = ["Event Title", "Start Time", "End Time", "Duration (Hrs)", "All Day?", "Location", "My Status"]; sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#eeeeee"); // Calculate the timeframe (Starting from right now) var start = new Date(); var end = new Date(); end.setDate(end.getDate() + days); // Access your personal Google Calendar and fetch all events in the range var calendar = CalendarApp.getDefaultCalendar(); var events = calendar.getEvents(start, end); var eventList = []; // Loop through every single event found for (var i = 0; i < events.length; i++) { var e = events[i]; var myStatus = ""; // Check if you accepted the meeting or if you are the organizer try { myStatus = e.getMyStatus(); } catch(err) { myStatus = "Owner"; } // Math: Convert the difference between start/end from milliseconds to hours var duration = e.isAllDayEvent() ? "24" : ((e.getEndTime() - e.getStartTime()) / (3600000)).toFixed(2); // Add this event's data to our temporary storage list eventList.push([ e.getTitle(), e.getStartTime(), e.getEndTime(), duration, e.isAllDayEvent(), e.getLocation(), myStatus ]); } // If we found events, paste the whole list into the sheet at once if (eventList.length > 0) { sheet.getRange(2, 1, eventList.length, eventList[0].length).setValues(eventList); SpreadsheetApp.getUi().alert('Success! Found ' + eventList.length + ' events.'); } else { SpreadsheetApp.getUi().alert('No events found for this timeframe.'); } }
